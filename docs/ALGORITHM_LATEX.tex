\begin{algorithm}[t]
\caption{Dual-Anchor Reward-Aware PPO Training}
\label{alg:dual_anchor_rappo}
\begin{algorithmic}[1]
\Require Image-report data $\mathcal{D}$, report corpus $\mathcal{M}$,
maximum evidence count $k$, policy $\pi_\theta$, reference policy
$\pi_{\mathrm{ref}}$
\Ensure Aligned report-generation policy $\pi_\theta$
\State Train image and text encoders with symmetric InfoNCE.
\State Index training reports in the shared embedding space.
\State Initialize $\pi_\theta$ with retrieval-conditioned NLL training.
\For{each PPO update}
  \State Retrieve adaptive evidence $E_i$ with leave-one-out filtering.
  \State Assemble prompt $P_i$ from visual semantics and $E_i$.
  \State Sample $\hat{Y}_i \sim \pi_\theta(\cdot \mid P_i)$.
  \State Compute image-grounded reward $r_{\mathrm{IFR}}$.
  \State Compute reference-guided reward $r_{\mathrm{RCR}}$.
  \State $r_{\mathrm{DAR}} \gets
  \lambda_I r_{\mathrm{IFR}} + \lambda_R r_{\mathrm{RCR}}$.
  \State Update $\pi_\theta$ with clipped PPO and KL regularization to
  $\pi_{\mathrm{ref}}$.
\EndFor
\State \Return $\pi_\theta$
\end{algorithmic}
\end{algorithm}
