\begin{algorithm}[t]
\caption{Conceptual Dual-Perspective Alignment for Medical Report Generation}
\label{alg:conceptual_alignment}
\begin{algorithmic}[1]
\Require Private medical image-report data, private evidence system, private report generator
\Ensure Conceptually aligned report generator
\State Build an internal evidence store from permitted private materials.
\For{each private training case}
  \State Collect relevant contextual evidence using the internal retrieval system.
  \State Generate a candidate report draft under a private context template.
  \State Assess the draft from a visual-grounding perspective.
  \State Assess the draft from a textual-coherence perspective.
  \State Combine the private feedback signals.
  \State Update the generator with the internal alignment procedure.
\EndFor
\State \Return aligned generator
\end{algorithmic}
\end{algorithm}
